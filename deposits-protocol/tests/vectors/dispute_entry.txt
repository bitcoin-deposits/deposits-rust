# broadcast carries an embedding?  The rule holds under every ruleset (DEP-18), so each case is
# stated under balance-anchor-v6 and grounded-arms-v7 with the same answer.
#   embedding: none (no embedding key) | placeholder (ledger_id = accused, sequence 0, field "inline",
#              empty update_hash: present but never in any history) | valid (the proof hash is at
#              the named sequence of the accused's history, empty causal chain)
#   requires: 1 if the type needs an embedding (DEP-06 §Embedding), else 0
#   entry: permitted | refused  (whether the rule lets a member enter; a member MAY still decline,
